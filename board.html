// ⚠️ 발급받으신 본인의 GAS 웹 앱 URL을 입력하세요. const GAS_WEB_APP_URL = 'https://script.google.com/macros/s/AKfycbyvNvNmTi2ZAECH3hrwNVsRZkIbIgF-HAA7GE3RcQjmGTSe4l15HytFD5qBBkd7kaha/exec'; window.onload = function() { loadMessages(); }; // 방명록 목록 불러오기 (GET) async function loadMessages() { const loading = document.getElementById('loading'); const listDiv = document.getElementById('messageList'); if (loading) loading.style.display = 'block'; try { // redirect: "follow" 옵션 추가 const response = await fetch(GAS_WEB_APP_URL, { redirect: "follow" }); const result = await response.json(); if (loading) loading.style.display = 'none'; listDiv.innerHTML = ''; if (result.status === 'success' && result.data && result.data.length > 0) { result.data.forEach(item => { const div = document.createElement('div'); div.className = 'msg-item'; div.innerHTML = `
${escapeHtml(item.name)} ${escapeHtml(item.date)}
${escapeHtml(item.message)}
`; listDiv.appendChild(div); }); } else { listDiv.innerHTML = '

작성된 방명록이 없습니다.

'; } } catch (err) { if (loading) loading.innerText = '목록을 불러오는 중 오류가 발생했습니다.'; console.error("불러오기 에러:", err); } } // 방명록 등록 제출 (POST) async function handleFormSubmit(event) { event.preventDefault(); const btn = document.getElementById('submitBtn'); const name = document.getElementById('name').value.trim(); const message = document.getElementById('message').value.trim(); if (!name || !message) return; btn.disabled = true; btn.innerText = '저장 중...'; try { // CORS 회피용 text/plain 헤더 설정 const response = await fetch(GAS_WEB_APP_URL, { method: 'POST', headers: { 'Content-Type': 'text/plain;charset=utf-8' }, body: JSON.stringify({ name: name, message: message }), redirect: "follow" }); const result = await response.json(); if (result.status === 'success') { document.getElementById('guestbookForm').reset(); loadMessages(); // 저장 성공 시 목록 다시 불러오기 } else { alert('저장 실패: ' + (result.message || '알 수 없는 오류')); } } catch (err) { console.error("제출 에러:", err); alert('전송 중 오류가 발생했습니다. 키보드 F12(개발자 도구) 콘솔을 확인해 보세요.'); } finally { btn.disabled = false; btn.innerText = '방명록 남기기'; } } function escapeHtml(text) { if (!text) return ''; return String(text) .replace(/&/g, "&") .replace(//g, ">") .replace(/"/g, """) .replace(/'/g, "'"); }
