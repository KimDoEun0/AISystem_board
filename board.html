const GAS_WEB_APP_URL = 'https://script.google.com/macros/s/AKfycbyvNvNmTi2ZAECH3hrwNVsRZkIbIgF-HAA7GE3RcQjmGTSe4l15HytFD5qBBkd7kaha/exec'; // 1. 방명록 목록 불러오기 (GET) async function loadMessages() { const loading = document.getElementById('loading'); const listDiv = document.getElementById('messageList'); if (loading) loading.style.display = 'block'; try { // redirect: "follow"를 명시하여 GAS의 302 리다이렉트를 정상 추적합니다. const response = await fetch(GAS_WEB_APP_URL, { redirect: "follow" }); const result = await response.json(); if (loading) loading.style.display = 'none'; listDiv.innerHTML = ''; if (result.status === 'success' && result.data.length > 0) { result.data.forEach(item => { const div = document.createElement('div'); div.className = 'msg-item'; div.innerHTML = `
${escapeHtml(item.name)} ${escapeHtml(item.date)}
${escapeHtml(item.message)}
`; listDiv.appendChild(div); }); } else { listDiv.innerHTML = '

작성된 방명록이 없습니다.

'; } } catch (err) { console.error("목록 불러오기 오류:", err); if (loading) loading.innerText = '목록을 불러오는 중 오류가 발생했습니다.'; } } // 2. 방명록 제출 (POST) async function handleFormSubmit(event) { event.preventDefault(); const btn = document.getElementById('submitBtn'); const name = document.getElementById('name').value.trim(); const message = document.getElementById('message').value.trim(); if (!name || !message) return; btn.disabled = true; btn.innerText = '저장 중...'; try { // CORS Preflight를 피하기 위해 text/plain 타입으로 포장하여 전송합니다. const response = await fetch(GAS_WEB_APP_URL, { method: 'POST', headers: { 'Content-Type': 'text/plain;charset=utf-8' }, body: JSON.stringify({ name: name, message: message }), redirect: "follow" }); const result = await response.json(); if (result.status === 'success') { document.getElementById('guestbookForm').reset(); loadMessages(); // 작성 후 목록 재갱신 } else { alert('저장 실패: ' + result.message); } } catch (err) { console.error("제출 오류:", err); alert('전송 중 오류가 발생했습니다. 브라우저 콘솔(F12)을 확인하세요.'); } finally { btn.disabled = false; btn.innerText = '방명록 남기기'; } } function escapeHtml(text) { if (!text) return ''; return String(text) .replace(/&/g, "&") .replace(//g, ">") .replace(/"/g, """) .replace(/'/g, "'"); } window.onload = loadMessages;
